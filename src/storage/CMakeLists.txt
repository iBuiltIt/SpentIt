if(NOT UNIX)
  message(FATAL_ERROR "Storage prototype uses POSIX exclusive publication and process tests; Windows is not implemented")
endif()
set(SPENTIT_SQLCIPHER_SOURCE "" CACHE PATH "Directory containing pinned generated SQLCipher sqlite3.c and sqlite3.h")
foreach(item IN ITEMS sqlite3.c sqlite3.h)
  if(NOT EXISTS "${SPENTIT_SQLCIPHER_SOURCE}/${item}")
    message(FATAL_ERROR "Generate pinned SQLCipher sources first; see doc/storage-prototype.md")
  endif()
endforeach()
file(SHA256 "${SPENTIT_SQLCIPHER_SOURCE}/sqlite3.c" sqlcipher_c_hash)
file(SHA256 "${SPENTIT_SQLCIPHER_SOURCE}/sqlite3.h" sqlcipher_h_hash)
if(NOT sqlcipher_c_hash STREQUAL "8640c653acadf665cce6331646f60b5b74a4690746f2c4a2d8f688a0570a0c0c" OR
   NOT sqlcipher_h_hash STREQUAL "8a9d1bff44d75174ca6dea3ea9bac50a6104d86facb566647b8bb839375b7b3a")
  message(FATAL_ERROR "SQLCipher generated sources do not match the reviewed pin")
endif()
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(spentit_sqlcipher STATIC "${SPENTIT_SQLCIPHER_SOURCE}/sqlite3.c")
target_include_directories(spentit_sqlcipher SYSTEM PUBLIC "${SPENTIT_SQLCIPHER_SOURCE}")
target_compile_definitions(spentit_sqlcipher PUBLIC SQLITE_HAS_CODEC PRIVATE
  SQLITE_TEMP_STORE=3 SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION
  SQLITE_EXTRA_INIT=sqlcipher_extra_init SQLITE_EXTRA_SHUTDOWN=sqlcipher_extra_shutdown)
target_link_libraries(spentit_sqlcipher PRIVATE OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
if(SPENTIT_SANITIZERS)
  target_compile_options(spentit_sqlcipher PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(spentit_sqlcipher PRIVATE -fsanitize=address,undefined)
endif()
add_library(spentit_storage_prototype STATIC prototype_store.cpp)
target_include_directories(spentit_storage_prototype PUBLIC include)
target_link_libraries(spentit_storage_prototype PUBLIC SpentIt::engine PRIVATE spentit_sqlcipher)
set_target_properties(spentit_storage_prototype PROPERTIES CXX_EXTENSIONS OFF)
spentit_compiler_options(spentit_storage_prototype)

if(SPENTIT_IMPORT_WORKFLOW)

target_include_directories(spentit_import_store PUBLIC include "${PROJECT_SOURCE_DIR}/src/app/include")
target_link_libraries(spentit_import_store PUBLIC SpentIt::engine PRIVATE spentit_sqlcipher)
if(SPENTIT_RULES)
  # New layout exception: the repository re-runs the one application matcher.

  target_compile_definitions(spentit_import_store PRIVATE SPENTIT_RULES_STORAGE)
  target_link_libraries(spentit_import_store PRIVATE SpentIt::rules spentit_rule_preview)
endif()
set_target_properties(spentit_import_store PROPERTIES CXX_EXTENSIONS OFF)
spentit_compiler_options(spentit_import_store)
endif()
