  if(WIN32)
    set(SPENTIT_PCRE2_ROOT "" CACHE PATH "Install prefix of PCRE2 10.49")
    find_path(PCRE2_INCLUDE_DIR NAMES pcre2.h HINTS "${SPENTIT_PCRE2_ROOT}/include" REQUIRED)
    find_library(PCRE2_LIBRARY NAMES pcre2-8 pcre2-8-static HINTS "${SPENTIT_PCRE2_ROOT}/lib" REQUIRED)
    file(READ "${PCRE2_INCLUDE_DIR}/pcre2.h" spentit_pcre2_header)
    if(NOT spentit_pcre2_header MATCHES "#define[ \t]+PCRE2_MAJOR[ \t]+10" OR
       NOT spentit_pcre2_header MATCHES "#define[ \t]+PCRE2_MINOR[ \t]+49")
      message(FATAL_ERROR "PCRE2 must be exactly 10.49")
    endif()
    if(spentit_pcre2_header MATCHES "#define[ \t]+PCRE2_PRERELEASE[ \t]+[0-9A-Za-z]")
      message(FATAL_ERROR "PCRE2 must be exactly 10.49")
    endif()
    add_library(SpentIt_PCRE2 STATIC IMPORTED)
    set_target_properties(SpentIt_PCRE2 PROPERTIES
      IMPORTED_LOCATION "${PCRE2_LIBRARY}"
      INTERFACE_INCLUDE_DIRECTORIES "${PCRE2_INCLUDE_DIR}"
      INTERFACE_COMPILE_DEFINITIONS "PCRE2_STATIC")
    set(SPENTIT_PCRE2_TARGET SpentIt_PCRE2)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PCRE2 REQUIRED IMPORTED_TARGET libpcre2-8=10.49)
    set(SPENTIT_PCRE2_TARGET PkgConfig::PCRE2)
  endif()
