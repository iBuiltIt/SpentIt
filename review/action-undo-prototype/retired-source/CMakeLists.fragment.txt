
# Explicitly opt in to this development measurement executable. It instruments
# real repository commands but is not the unified Undo implementation.
option(SPENTIT_ACTION_JOURNAL_PROBE "Build the development action-journal measurement probe" OFF)
if(SPENTIT_IMPORT_WORKFLOW AND SPENTIT_ACTION_JOURNAL_PROBE)
  add_executable(spentit_action_journal_probe action_journal_probe.cpp)
  target_link_libraries(spentit_action_journal_probe PRIVATE spentit_import_store spentit_sqlcipher)
  spentit_compiler_options(spentit_action_journal_probe)
endif()

if(SPENTIT_IMPORT_WORKFLOW AND SPENTIT_ACTION_JOURNAL_PROBE)
  # Compile the actual store in a separate test library. No prototype SQL
  # translation or schema shape enters the application library.
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  add_library(spentit_action_payload_store STATIC
    action_payload_store.cpp
    ../../src/storage/document_gateway.cpp
    ../../src/storage/verification_cache.cpp
    ../../src/storage/document_lock.cpp)
  get_target_property(action_probe_links spentit_import_store LINK_LIBRARIES)
  get_target_property(action_probe_includes spentit_import_store INCLUDE_DIRECTORIES)
  target_link_libraries(spentit_action_payload_store PRIVATE ${action_probe_links} PUBLIC SpentIt::engine spentit_diagnostics)
  target_include_directories(spentit_action_payload_store PUBLIC ${action_probe_includes})
  if(SPENTIT_RULES)
    target_compile_definitions(spentit_action_payload_store PRIVATE SPENTIT_RULES_STORAGE)
  endif()
  spentit_compiler_options(spentit_action_payload_store)
  add_executable(spentit_action_payload_probe action_journal_probe.cpp)
  target_compile_definitions(spentit_action_payload_probe PRIVATE SPENTIT_ACTION_PAYLOAD_PROBE)
  target_link_libraries(spentit_action_payload_probe PRIVATE spentit_action_payload_store spentit_sqlcipher)
  spentit_compiler_options(spentit_action_payload_probe)
endif()
