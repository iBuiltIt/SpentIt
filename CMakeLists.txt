cmake_minimum_required(VERSION 3.21)
project(SpentIt VERSION 0.1.0 LANGUAGES CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build (see doc/building.md)")
endif()
include(CTest)
option(SPENTIT_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(cmake/CompilerOptions.cmake)
add_subdirectory(src/engine)
option(SPENTIT_IMPORT_WORKFLOW "Build the experimental durable QIF workflow" OFF)
option(SPENTIT_STORAGE_PROTOTYPE "Build the experimental SQLCipher storage and recovery tests" OFF)
if(SPENTIT_IMPORT_WORKFLOW AND NOT SPENTIT_STORAGE_PROTOTYPE)
  message(FATAL_ERROR "Durable import requires SPENTIT_STORAGE_PROTOTYPE for pinned SQLCipher")
endif()
if(SPENTIT_STORAGE_PROTOTYPE)
  enable_language(C)
  add_subdirectory(src/storage)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests/engine)
  if(SPENTIT_STORAGE_PROTOTYPE)
    add_subdirectory(tests/storage)
  endif()
endif()

option(SPENTIT_SHEET_PROTOTYPE "Build the synthetic Qt Quick sheet experiment" OFF)
if(SPENTIT_SHEET_PROTOTYPE OR SPENTIT_IMPORT_WORKFLOW)
  add_subdirectory(src/app)
endif()
if(SPENTIT_SHEET_PROTOTYPE)
  add_subdirectory(src/ui)
endif()
