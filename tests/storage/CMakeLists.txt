# Each operation has its own result and runs its complete write/sync sweep.
function(spentit_fault_scenarios name target mode last timeout)
  foreach(scenario RANGE 0 ${last})
    math(EXPR end "${scenario} + 1")
    add_test(NAME ${name}.${scenario} COMMAND ${target} ${mode} ${scenario} ${end})
    set_tests_properties(${name}.${scenario} PROPERTIES LABELS durability TIMEOUT ${timeout})
    if(SPENTIT_SANITIZERS AND mode STREQUAL "crashes")
      set_tests_properties(${name}.${scenario} PROPERTIES ENVIRONMENT "ASAN_OPTIONS=quarantine_size_mb=8:malloc_context_size=0")
    endif()
  endforeach()
endfunction()
